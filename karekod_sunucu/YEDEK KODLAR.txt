from pynput.keyboard import Key, Controller
keyboard = Controller()

# --- GLOBAL AYARLAR ---
SERVER_CONFIG_FILE = "sunucu_ayarlar.json"
TARGET_BASE_PATH = os.path.join(os.path.expanduser('~'), 'Desktop')
QR_PACKAGE_PATH = os.path.join(os.path.expanduser('~'), 'Desktop')
ui_queue = Queue()
flask_thread = None
tray_icon = None

def resource_path(relative_path):
    try:
        base_path = sys._MEIPASS
    except Exception:
        base_path = os.path.abspath(".")
    return os.path.join(base_path, relative_path)

app = Flask(__name__)

# --- YENÝ YAZMA FONKSÝYONU (PYNPUT - DAHA GÜÇLÜ) ---
def type_str_robust(text):
    """
    Medula/Tebos için en güçlü yazma yöntemi.
    Tuþlara tek tek fiziksel olarak basýlýp çekilmesini simüle eder.
    """
    try:
        for char in text:
            keyboard.type(char)
            time.sleep(0.005) # Çok kýsa bekleme (harfler karýþmasýn diye)
        
        time.sleep(0.05)
        keyboard.press(Key.enter)
        keyboard.release(Key.enter)
    except Exception as e:
        ui_queue.put(('log', f"Klavye Hatasý: {e}"))

@app.route('/send', methods=['POST'])
def receive_barcode():
    try:
        data = request.json
        barcodes = data.get('barcodes', [])
        
        # UI GÜNCELLEME (ANINDA)
        ui_queue.put(("STATUS", {'text': 'VERÝ GELDÝ - YAZILIYOR...', 'color': 'green'}))

        items_to_process = [item for item in barcodes if isinstance(item, str) and item.strip()]
        
        if not items_to_process:
            return jsonify({"status": "empty"}), 200

        for barcode in items_to_process:
            type_str_robust(barcode)
            
            # PC HOPARLÖRÜNDEN BÝP SESÝ
            try:
                winsound.Beep(600, 100)
            except: pass
            
            time.sleep(0.3) # Ýki karekod arasý bekleme

        ui_queue.put(('log', f"{len(items_to_process)} adet aktarýldý."))
        # Ýþlem bitince tekrar Hazýr moda dön
        ui_queue.put(("STATUS", {'text': 'BAÐLANTI AKTÝF - HAZIR', 'color': '#28a745'}))
        return jsonify({"status": "ok"}), 200

    except Exception as e:
        ui_queue.put(('log', f"Hata: {e}"))
        return jsonify({"status": "error", "message": str(e)}), 500

@app.route('/package', methods=['POST'])
def receive_package():
    ui_queue.put(("STATUS", {'text': 'PAKET AÇILIYOR...', 'color': 'blue'}))
    try:
        data = request.json
        qr_name = secure_filename(data.get('name', 'Paket'))
        barcodes_in_package = data.get('barcodes', [])
        folder_name = data.get('folder_name', 'QR Paketleri')

        for barcode in barcodes_in_package:
            type_str_robust(barcode)
            time.sleep(0.2)

        # QR Resmini Kaydet
        try:
            full_data_string = json.dumps(barcodes_in_package)
            qr = qrcode.QRCode(version=1, box_size=10, border=4)
            qr.add_data(full_data_string)
            qr.make(fit=True)
            qr_img = qr.make_image(fill_color="black", back_color="white")
            
            target_folder = os.path.join(QR_PACKAGE_PATH, folder_name)
            os.makedirs(target_folder, exist_ok=True)
            file_path = os.path.join(target_folder, f"{qr_name}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.png")
            qr_img.save(file_path)
            ui_queue.put(('log', f"Paket Kaydedildi: {qr_name}"))
        except Exception as e:
            ui_queue.put(('log', f"Resim Hatasý: {e}"))

        ui_queue.put(("STATUS", {'text': 'BAÐLANTI AKTÝF', 'color': '#28a745'}))
        return jsonify({"status": "ok"}), 200
    except Exception as e:
        return jsonify({"status": "error", "message": str(e)}), 500

@app.route('/upload', methods=['POST'])
def upload_image():
    ui_queue.put(("STATUS", {'text': 'RESÝM ALINIYOR...', 'color': 'purple'}))
    if 'image' not in request.files: return jsonify({"status": "error"}), 400
    file = request.files['image']
    folder_name = request.form.get('folder_name', 'Gelen Resimler')
    if file:
        target_folder = os.path.join(TARGET_BASE_PATH, folder_name)
        os.makedirs(target_folder, exist_ok=True)
        filename = f"{datetime.now().strftime('%Y-%m-%d')}_{secure_filename(file.filename)}"
        file.save(os.path.join(target_folder, filename))
        ui_queue.put(('log', f"Resim: {filename}"))
        ui_queue.put(("STATUS", {'text': 'BAÐLANTI AKTÝF', 'color': '#28a745'}))
        return jsonify({"status": "ok"}), 200
    return jsonify({"status": "error"}), 500

def get_local_ip():
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        s.connect(('10.255.255.255', 1))
        IP = s.getsockname()[0]
    except:
        IP = '127.0.0.1'
    finally:
        s.close()
    return IP

def generate_qr_code(ip, port=5000):
    return qrcode.make(f"{ip}:{port}")

def save_settings():
    settings = {'image_path': TARGET_BASE_PATH, 'qr_package_path': QR_PACKAGE_PATH}
    with open(SERVER_CONFIG_FILE, 'w', encoding='utf-8') as f:
        json.dump(settings, f, indent=4)

def load_settings():
    global TARGET_BASE_PATH, QR_PACKAGE_PATH
    if os.path.exists(SERVER_CONFIG_FILE):
        try:
            with open(SERVER_CONFIG_FILE, 'r', encoding='utf-8') as f:
                settings = json.load(f)
                TARGET_BASE_PATH = settings.get('image_path', TARGET_BASE_PATH)
                QR_PACKAGE_PATH = settings.get('qr_package_path', QR_PACKAGE_PATH)
        except: pass

class ControlPanelApp:
    def __init__(self, root):
        self.root = root
        self.root.title("M-ONUR VERÝ SUNUCUSU (Yönetici)")
        self.root.geometry("950x750")
        
        # ÇARPIYA BASINCA GÝZLE (TEPSÝYE AT)
        self.root.protocol('WM_DELETE_WINDOW', self.hide_window)

        try:
            icon_path = resource_path("icon.png")
            if os.path.exists(icon_path):
                img = tk.PhotoImage(file=icon_path)
                self.root.iconphoto(False, img)
        except: pass

        style = ttk.Style()
        style.configure("TButton", font=("Arial", 10, "bold"), padding=10)
        style.configure("Header.TLabel", font=("Arial", 12, "bold"))
        
        main_frame = ttk.Frame(root, padding="15")
        main_frame.pack(fill="both", expand=True)
        
        # ÜST BÝLGÝ ALANI
        top_frame = ttk.LabelFrame(main_frame, text="Baðlantý Durumu", padding="10")
        top_frame.pack(fill="x", pady=(0,15))
        
        qr_frame = ttk.Frame(top_frame)
        qr_frame.pack(side="left", padx=10)
        self.qr_label = ttk.Label(qr_frame)
        self.qr_label.pack()
        
        info_frame = ttk.Frame(top_frame)
        info_frame.pack(side="left", fill="both", expand=True, padx=20)

        self.ip_label = ttk.Label(info_frame, text=f"IP: {get_local_ip()}:5000", font=("Arial", 20, "bold"), foreground="#333")
        self.ip_label.pack(anchor="w", pady=5)
        
        self.status_label = ttk.Label(info_frame, text="MOBÝL UYGULAMA BEKLENÝYOR...", foreground="red", font=("Arial", 14, "bold"))
        self.status_label.pack(anchor="w", pady=10)

        # TEST BUTONU (Sorununuzu çözmek için ekledim)
        btn_test = ttk.Button(info_frame, text="?? YAZMA TESTÝ YAP", command=self.run_write_test)
        btn_test.pack(anchor="w", pady=5)
        
        # AYARLAR ALANI
        settings_frame = ttk.LabelFrame(main_frame, text="Klasör Ayarlarý", padding="10")
        settings_frame.pack(fill="x", pady=(0,15))
        
        ttk.Button(settings_frame, text="Resim Klasörü", command=self.change_image_directory).pack(side="left", padx=5)
        self.path_lbl = ttk.Label(settings_frame, text=f"{TARGET_BASE_PATH}", font=("Arial", 9))
        self.path_lbl.pack(side="left", padx=5)
        
        # LOG ALANI
        bottom_frame = ttk.LabelFrame(main_frame, text="Ýþlem Geçmiþi", padding="10")
        bottom_frame.pack(fill="both", expand=True)

        self.log_list = Listbox(bottom_frame, height=10, font=("Consolas", 11), bg="#f0f0f0")
        self.log_list.pack(fill="both", expand=True, side="left")
        
        scroll = ttk.Scrollbar(bottom_frame, command=self.log_list.yview)
        scroll.pack(side="right", fill="y")
        self.log_list.config(yscrollcommand=scroll.set)
        
        ttk.Button(main_frame, text="Temizle", command=lambda: self.log_list.delete(0, 'end')).pack(pady=5, anchor="e")

        self.display_qr_code()
        self.process_queue()

    def process_queue(self):
        try:
            while not ui_queue.empty():
                msg_type, data = ui_queue.get_nowait()
                if msg_type == "STATUS":
                    self.status_label.config(text=data['text'], foreground=data['color'])
                    if data['color'] in ['green', '#28a745']:
                        self.ip_label.config(foreground="green")
                elif msg_type == "log":
                    t = datetime.now().strftime('%H:%M:%S')
                    self.log_list.insert(0, f"[{t}] {data}")
        except: pass
        self.root.after(200, self.process_queue)

    def display_qr_code(self):
        qr_img = generate_qr_code(get_local_ip()).resize((160, 160))
        self.tk_qr = ImageTk.PhotoImage(qr_img)
        self.qr_label.config(image=self.tk_qr)

    def change_image_directory(self):
        path = filedialog.askdirectory()
        if path:
            global TARGET_BASE_PATH
            TARGET_BASE_PATH = path
            self.path_lbl.config(text=path)
            save_settings()

    def run_write_test(self):
        messagebox.showinfo("Test", "Tamam'a bastýktan sonra 3 saniye içinde bir Not Defterine veya Yazý alanýna týklayýn.")
        time.sleep(3)
        type_str_robust("Test Basarili 12345")

    def hide_window(self):
        """Pencereyi gizler (System Tray'e gönderir)"""
        self.root.withdraw()

def start_flask():
    app.run(host='0.0.0.0', port=5000, use_reloader=False)

def setup_tray(root_window):
    def show_window(icon, item):
        root_window.after(0, root_window.deiconify)

    def exit_app(icon, item):
        icon.stop()
        root_window.after(0, root_window.destroy)
        os._exit(0)

    try:
        image = Image.open(resource_path("icon.png"))
    except:
        image = Image.new('RGB', (64, 64), color = (0, 120, 215))

    menu = pystray.Menu(
        pystray.MenuItem("Kontrol Panelini Aç", show_window, default=True),
        pystray.MenuItem("Çýkýþ", exit_app)
    )
    return pystray.Icon("M-ONUR", image, "M-ONUR Sunucu", menu)

if __name__ == '__main__':
    load_settings()
    flask_thread = threading.Thread(target=start_flask, daemon=True)
    flask_thread.start()
    
    root = tk.Tk()
    app_gui = ControlPanelApp(root)
    
    tray_icon = setup_tray(root)
    threading.Thread(target=tray_icon.run, daemon=True).start()
    
    root.mainloop()
